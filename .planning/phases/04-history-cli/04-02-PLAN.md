data Command
  = Generate GenerateOpts
  | Simulate SimulateOpts

  { simYears :: Int
  , simWorldFile :: FilePath  -- load serialized world

data ExportOpts = ExportOpts
  { exportFormat :: Format

  }

**JSON.hs** - Thin wrapper around Aeson (already used in 04-01):
```haskell
module Axiom.Export.JSON where

import Data.Aeson (encode, decode)
import qualified Data.ByteString.Lazy as BL

exportWorldJSON :: WorldState -> FilePath -> IO ()
exportWorldJSON world path = BL.writeFile path (encode world)

importWorldJSON :: FilePath -> IO (Maybe WorldState)
importWorldJSON path = decode <$> BL.readFile path
```

No custom ToJSON instances needed - WorldState already has Generic derivation from 04-01. This is just a convenience wrapper.

**ASCII.hs** - Render world state as ASCII map:
```haskell
module Axiom.Export.ASCII where

import Data.Text (Text)
import qualified Data.Text as T
import qualified Data.Text.IO as TIO

renderASCII :: WorldState -> Text
renderASCII world = T.unlines $ map renderRow rows
  where
    rows = -- grid representation from world
    renderRow = T.concat . map tileChar
    tileChar tile = case tileBiome tile of
      Desert -> "."
      Forest -> "T"
      Ocean  -> "~"
      Tundra -> "^"
      -- Add river markers, elevation indicators, etc.

exportWorldASCII :: WorldState -> FilePath -> IO ()
exportWorldASCII world path = TIO.writeFile path (renderASCII world)
```

Keep ASCII simple - use single characters for biomes, optionally show elevation with characters (lowlands='.', hills='=', mountains='^'). Add legend at bottom.
